class Solution {
    int middle(int a, int b, int c) {
        // code here
        if(a>b){
            if(b>c) return b;
            else if(c>a) return a;
            else return c;
        }else{
            if(a>c) return a;
            else if(c>b) return b;
            else return c;
        }
    }
}
